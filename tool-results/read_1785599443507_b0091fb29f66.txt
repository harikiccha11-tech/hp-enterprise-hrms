   133→function initials(name: string): string {
   134→  return name
   135→    .split(' ')
   136→    .filter(Boolean)
   137→    .slice(0, 2)
   138→    .map((w) => w[0]?.toUpperCase())
   139→    .join('')
   140→}
   141→
   142→function formatINR(amount: number): string {
   143→  return new Intl.NumberFormat('en-IN', {
   144→    style: 'currency',
   145→    currency: 'INR',
   146→    maximumFractionDigits: 0,
   147→  }).format(amount)
   148→}
   149→
   150→function fmtDate(d: string | null | undefined): string {
   151→  if (!d) return '—'
   152→  try {
   153→    return format(new Date(d), 'dd MMM yyyy')
   154→  } catch {
   155→    return '—'
   156→  }
   157→}
   158→
   159→function statusColor(status: string): string {
   160→  const s = status.toUpperCase()
   161→  if (s === 'ACTIVE' || s === 'OPEN' || s === 'PAID') return 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 border-emerald-500/25'
   162→  if (s === 'COMPLETED' || s === 'SENT') return 'bg-sky-500/15 text-sky-700 dark:text-sky-400 border-sky-500/25'
   163→  if (s === 'ON_HOLD' || s === 'DRAFT') return 'bg-amber-500/15 text-amber-700 dark:text-amber-400 border-amber-500/25'
   164→  if (s === 'REJECTED' || s === 'OVERDUE') return 'bg-red-500/15 text-red-700 dark:text-red-400 border-red-500/25'
   165→  if (s === 'CLOSED') return 'bg-slate-500/15 text-slate-600 dark:text-slate-400 border-slate-500/25'
   166→  return 'bg-primary/10 text-primary border-primary/20'
   167→}

   218→function DataTable<T extends { id: string }>({
   219→  title,
   220→  data,
   221→  loading,
   222→  emptyMessage,
   223→  emptyIcon: EmptyIcon,
   224→  columns,
   225→  renderRow,
   226→}: {
   227→  title: string
   228→  data: T[]
   229→  loading: boolean
   230→  emptyMessage: string
   231→  emptyIcon: typeof LayoutDashboard
   232→  columns: { key: string; label: string; className?: string }[]
   233→  renderRow: (item: T) => React.ReactNode
   234→}) {
   235→  return (
   236→    <Card>
   237→      <CardHeader className="pb-3">
   238→        <CardTitle className="text-base">{title}</CardTitle>
   239→      </CardHeader>
   240→      <CardContent className="px-6 pb-6 pt-0">
   241→        {loading ? (
   242→          <div className="space-y-3">
   243→            {Array.from({ length: 4 }).map((_, i) => (
   244→              <Skeleton key={i} className="h-10 w-full" />
   245→            ))}
   246→          </div>
   247→        ) : data.length === 0 ? (
   248→          <div className="flex flex-col items-center justify-center py-10 text-center text-muted-foreground">
   249→            <EmptyIcon className="mb-2 h-8 w-8 opacity-40" />
   250→            <p className="text-sm font-medium">{emptyMessage}</p>
   251→            <p className="mt-0.5 text-xs">Data will appear here once available</p>
   252→          </div>
   253→        ) : (
   254→          <div className="max-h-96 overflow-y-auto rounded-lg border scroll-thin">
   255→            <Table>
   256→              <TableHeader className="bg-muted/50">
   257→                <TableRow>
   258→                  {columns.map((col) => (
   259→                    <TableHead key={col.key} className={cn('text-xs font-semibold uppercase tracking-wide text-muted-foreground', col.className)}>
   260→                      {col.label}
   261→                    </TableHead>
   262→                  ))}
   263→                </TableRow>
   264→              </TableHeader>
   265→              <TableBody>
   266→                {data.map((item) => renderRow(item))}
   267→              </TableBody>
   268→            </Table>
   269→          </div>
   270→        )}
   271→      </CardContent>
   272→    </Card>
   273→  )
   274→}

   577→          <p className="text-xs uppercase tracking-wider text-muted-foreground">Paid</p>

   582→          <p className="text-xs uppercase tracking-wider text-muted-foreground">Outstanding</p>

   584→          <p className="mt-0.5 text-[11px] text-muted-foreground">Pending / overdue</p>

   586→      </div>

   591→        loading={loading}

   633→  const [unread, setUnread] = useState(0)

   748→            </button>

   751→      </nav>

   753→      {/* User card + logout */}
   754→      <div className="border-t border-white/10 p-3">
   755→        <div className="rounded-lg bg-white/5 p-3">

   792→
