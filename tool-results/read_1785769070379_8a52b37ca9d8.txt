   269→function Reveal({ children, className, delay = 0, direction = 'up' }: {
   270→  children: React.ReactNode; className?: string; delay?: number; direction?: 'up' | 'down' | 'left' | 'right'
   271→}) {

   277→    <motion.div
   278→      ref={ref}
   279→      initial={{ opacity: 0, ...dir }}
   280→      animate={inView ? { opacity: 1, x: 0, y: 0 } : {}}
   281→      transition={{ duration: 0.6, delay, ease: [0.25, 0.4, 0.25, 1] }}
   282→      className={className}
   283→    >
   284→      {children}
   285→    </motion.div>
   286→  )
   287→}
