   134→function initials(name: string): string {
   135→  return name
   136→    .split(' ')
   137→    .filter(Boolean)
   138→    .slice(0, 2)
   139→    .map((w) => w[0]?.toUpperCase())
   140→    .join('')
   141→}

   143→function formatINR(amount: number): string {
   144→  return new Intl.NumberFormat('en-IN', {
   145→    style: 'currency',
   146→    currency: 'INR',
   147→    maximumFractionDigits: 0,
   148→  }).format(amount)
   149→}
   150→
   151→function fmtDate(d: string | null | undefined): string {
   152→  if (!d) return '—'
   153→  try {
   154→    return format(new Date(d), 'dd MMM yyyy')
   155→  } catch {
   156→    return '—'
   157→  }
   158→}
   159→
   160→function statusColor(status: string): string {
   161→  const s = status.toUpperCase()
   162→  if (s === 'ACTIVE' || s === 'OPEN' || s === 'PAID') return 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 border-emerald-500/25'
   163→  if (s === 'COMPLETED' || s === 'SENT') return 'bg-sky-500/15 text-sky-700 dark:text-sky-400 border-sky-500/25'
   164→  if (s === 'ON_HOLD' || s === 'DRAFT') return 'bg-amber-500/15 text-amber-700 dark:text-amber-400 border-amber-500/25'
   165→  if (s === 'REJECTED' || s === 'OVERDUE') return 'bg-red-500/15 text-red-700 dark:text-red-400 border-red-500/25'
   166→  if (s === 'CLOSED') return 'bg-slate-500/15 text-slate-600 dark:text-slate-400 border-slate-500/25'
   167→  return 'bg-primary/10 text-primary border-primary/20'
   168→}

   219→function DataTable<T extends { id: string }>({
   220→  title,
   221→  data,
   222→  loading,
   223→  emptyMessage,
   224→  emptyIcon: EmptyIcon,
   225→  columns,
   226→  renderRow,
   227→}: {
   228→  title: string
   229→  data: T[]
   230→  loading: boolean
   231→  emptyMessage: string
   232→  emptyIcon: typeof LayoutDashboard
   233→  columns: { key: string; label: string; className?: string }[]
   234→  renderRow: (item: T) => React.ReactNode
   235→}) {
   236→  return (
   237→    <Card>
   238→      <CardHeader className="pb-3">
   239→        <CardTitle className="text-base">{title}</CardTitle>
   240→      </CardHeader>
   241→      <CardContent className="px-6 pb-6 pt-0">
   242→        {loading ? (
   243→          <div className="space-y-3">
   244→            {Array.from({ length: 4 }).map((_, i) => (
   245→              <Skeleton key={i} className="h-10 w-full" />
   246→            ))}
   247→          </div>
   248→        ) : data.length === 0 ? (
   249→          <div className="flex flex-col items-center justify-center py-10 text-center text-muted-foreground">
   250→            <EmptyIcon className="mb-2 h-8 w-8 opacity-40" />
   251→            <p className="text-sm font-medium">{emptyMessage}</p>
   252→            <p className="mt-0.5 text-xs">Data will appear here once available</p>
   253→          </div>
   254→        ) : (
   255→          <div className="max-h-96 overflow-y-auto rounded-lg border scroll-thin">
   256→            <Table>
   257→              <TableHeader className="bg-muted/50">
   258→                <TableRow>
   259→                  {columns.map((col) => (
   260→                    <TableHead key={col.key} className={cn('text-xs font-semibold uppercase tracking-wide text-muted-foreground', col.className)}>
   261→                      {col.label}
   262→                    </TableHead>
   263→                  ))}
   264→                </TableRow>
   265→              </TableHeader>
   266→              <TableBody>
   267→                {data.map((item) => renderRow(item))}
   268→              </TableBody>
   269→            </Table>
   270→          </div>
   271→        )}
   272→      </CardContent>
   273→    </Card>
   274→  )
   275→}

   563→function InvoicesView({ data, loading }: { data: DashboardData | null; loading: boolean }) {

   565→  const totalAmount = list.reduce((s, i) => s + i.total, 0)

   567→
   568→  return (
   569→    <div className="space-y-5">
   570→      {/* Summary Cards */}

   576→        </Card>
   577→        <Card className="p-4">
   578→          <p className="text-xs uppercase tracking-wider text-muted-foreground">Paid</p>
   579→          <p className="mt-1 text-xl font-bold text-emerald-700">{formatINR(totalPaid)}</p>
   580→          <p className="mt-0.5 text-[11px] text-muted-foreground">PAID status</p>
   581→        </Card>
   582→        <Card className="p-4">
   583→          <p className="text-xs uppercase tracking-wider text-muted-foreground">Outstanding</p>
   584→          <p className="mt-1 text-xl font-bold text-amber-700">{formatINR(totalAmount - totalPaid)}</p>
   585→          <p className="mt-0.5 text-[11px] text-muted-foreground">Pending / overdue</p>
   586→        </Card>
   587→      </div>
   588→
   589→      <DataTable
   590→        title="All Invoices"
   591→        data={list}
   592→        loading={loading}
   593→        emptyMessage="No invoices found"
   594→        emptyIcon={FileText}

   597→          { key: 'workOrder', label: 'Work Order', className: 'hidden md:table-cell' },
   598→          { key: 'amount', label: 'Amount', className: 'text-right' },

   607→            <TableCell className="hidden md:table-cell text-sm text-muted-foreground">{inv.workOrderTitle || '—'}</TableCell>
   608→            <TableCell className="text-right font-mono">{formatINR(inv.amount)}</TableCell>
   609→            <TableCell className="hidden lg:table-cell text-right font-mono text-muted-foreground">{formatINR(inv.tax)}</TableCell>
   610→            <TableCell className="text-right font-mono font-bold text-[var(--navy)] dark:text-white">{formatINR(inv.total)}</TableCell>
