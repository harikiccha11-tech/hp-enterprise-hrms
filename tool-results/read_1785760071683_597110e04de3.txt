   192→  const [password, setPassword] = useState('')

   194→  const [forgotOpen, setForgotOpen] = useState(false)
   195→  const Icon = portal.icon
   196→
   197→  const handleSubmit = async (e: React.FormEvent) => {
   198→    e.preventDefault()

   200→      toast.error('Please enter username and password')
   201→      return

   203→    setLoading(true)

   205→      const res = await fetch('/api/auth/login', {
   206→        method: 'POST',
   207→        headers: { 'Content-Type': 'application/json' },
   208→        body: JSON.stringify({ username: username.trim(), password }),
   209→      })
   210→      const data = await res.json()
   211→      if (res.ok && data.user) {
   212→        setUser(data.user)
   213→        toast.success('Welcome back!')

   215→        toast.error(data.error || 'Login failed. Please check your credentials.')

   218→      toast.error('Network error. Please try again.')

   220→      setLoading(false)
