   143→   143→function formatINR(amount: number): string {
   144→   144→  return new Intl.NumberFormat('en-IN', {
   145→   145→    style: 'currency',
   146→   146→    currency: 'INR',

   151→   151→function fmtDate(d: string | null | undefined): string {
   152→   152→  if (!d) return '—'
   153→   153→  try {
   154→   154→    return format(new Date(d), 'dd MMM yyyy')
   155→   155→  } catch {
   156→   156→    return '—'
   157→   157→  }

   160→   160→function statusColor(status: string): string {
   161→   161→  const s = status.toUpperCase()
   162→   162→  if (s === 'ACTIVE' || s === 'OPEN' || s === 'PAID') return 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 border-emerald-500/25'
   163→   163→  if (s === 'COMPLETED' || s === 'SENT') return 'bg-sky-500/15 text-sky-700 dark:text-sky-400 border-sky-500/25'
   164→   164→  if (s === 'ON_HOLD' || s === 'DRAFT') return 'bg-amber-500/15 text-amber-700 dark:text-amber-400 border-amber-500/25'
   165→   165→  if (s === 'REJECTED' || s === 'OVERDUE') return 'bg-red-500/15 text-red-700 dark:text-red-400 border-red-500/25'
   166→   166→  if (s === 'CLOSED') return 'bg-slate-500/15 text-slate-600 dark:text-slate-400 border-slate-500/25'
   167→   167→  return 'bg-primary/10 text-primary border-primary/20'
   168→   168→}
