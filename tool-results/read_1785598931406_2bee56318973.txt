   132→function initials(name: string): string {
   133→  return name
   134→    .split(' ')
   135→    .filter(Boolean)
   136→    .slice(0, 2)
   137→    .map((w) => w[0]?.toUpperCase())
   138→    .join('')
   139→}

   141→function formatINR(amount: number): string {
   142→  return new Intl.NumberFormat('en-IN', {
   143→    style: 'currency',
   144→    currency: 'INR',
   145→    maximumFractionDigits: 0,
   146→  }).format(amount)
   147→}
   148→
   149→function fmtDate(d: string | null | undefined): string {
   150→  if (!d) return '—'
   151→  try {
   152→    return format(new Date(d), 'dd MMM yyyy')
   153→  } catch {
   154→    return '—'
   155→  }
   156→}
   157→
   158→function statusColor(status: string): string {
   159→  const s = status.toUpperCase()
   160→  if (s === 'ACTIVE' || s === 'OPEN' || s === 'PAID') return 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 border-emerald-500/25'
   161→  if (s === 'COMPLETED' || s === 'SENT') return 'bg-sky-500/15 text-sky-700 dark:text-sky-400 border-sky-500/25'
   162→  if (s === 'ON_HOLD' || s === 'DRAFT') return 'bg-amber-500/15 text-amber-700 dark:text-amber-400 border-amber-500/25'
   163→  if (s === 'REJECTED' || s === 'OVERDUE') return 'bg-red-500/15 text-red-700 dark:text-red-400 border-red-500/25'
   164→  if (s === 'CLOSED') return 'bg-slate-500/15 text-slate-600 dark:text-slate-400 border-slate-500/25'
   165→  return 'bg-primary/10 text-primary border-primary/20'
   166→}

   217→function DataTable<T extends { id: string }>({
   218→  title,
   219→  data,
   220→  loading,
   221→  emptyMessage,
   222→  emptyIcon: EmptyIcon,
   223→  columns,
   224→  renderRow,
   225→}: {
   226→  title: string
   227→  data: T[]
   228→  loading: boolean
   229→  emptyMessage: string
   230→  emptyIcon: typeof LayoutDashboard
   231→  columns: { key: string; label: string; className?: string }[]
   232→  renderRow: (item: T) => React.ReactNode
   233→}) {
   234→  return (
   235→    <Card>
   236→      <CardHeader className="pb-3">
   237→        <CardTitle className="text-base">{title}</CardTitle>
   238→      </CardHeader>
   239→      <CardContent className="px-6 pb-6 pt-0">
   240→        {loading ? (
   241→          <div className="space-y-3">
   242→            {Array.from({ length: 4 }).map((_, i) => (
   243→              <Skeleton key={i} className="h-10 w-full" />
   244→            ))}
   245→          </div>
   246→        ) : data.length === 0 ? (
   247→          <div className="flex flex-col items-center justify-center py-10 text-center text-muted-foreground">
   248→            <EmptyIcon className="mb-2 h-8 w-8 opacity-40" />
   249→            <p className="text-sm font-medium">{emptyMessage}</p>
   250→            <p className="mt-0.5 text-xs">Data will appear here once available</p>
   251→          </div>
   252→        ) : (
   253→          <div className="max-h-96 overflow-y-auto rounded-lg border scroll-thin">
   254→            <Table>
   255→              <TableHeader className="bg-muted/50">
   256→                <TableRow>
   257→                  {columns.map((col) => (
   258→                    <TableHead key={col.key} className={cn('text-xs font-semibold uppercase tracking-wide text-muted-foreground', col.className)}>
   259→                      {col.label}
   260→                    </TableHead>
   261→                  ))}
   262→                </TableRow>
   263→              </TableHeader>
   264→              <TableBody>
   265→                {data.map((item) => renderRow(item))}
   266→              </TableBody>
   267→            </Table>
   268→          </div>
   269→        )}
   270→      </CardContent>
   271→    </Card>
   272→  )
   273→}

   561→function InvoicesView({ data, loading }: { data: DashboardData | null; loading: boolean }) {

   565→
   566→  return (
   567→    <div className="space-y-5">
   568→      {/* Summary Cards */}

   574→        </Card>
   575→        <Card className="p-4">
   576→          <p className="text-xs uppercase tracking-wider text-muted-foreground">Paid</p>

   581→          <p className="text-xs uppercase tracking-wider text-muted-foreground">Outstanding</p>
   582→          <p className="mt-1 text-xl font-bold text-amber-700">{formatINR(totalAmount - totalPaid)}</p>
   583→          <p className="mt-0.5 text-[11px] text-muted-foreground">Pending / overdue</p>
   584→        </Card>
   585→      </div>
   586→
   587→      <DataTable
   588→        title="All Invoices"
   589→        data={list}
   590→        loading={loading}
   591→        emptyMessage="No invoices found"
   592→        emptyIcon={FileText}
   593→        columns={[
   594→          { key: 'number', label: 'Invoice #' },
   595→          { key: 'workOrder', label: 'Work Order', className: 'hidden md:table-cell' },

   603→          <TableRow key={inv.id} className="hover:bg-muted/40">

   605→            <TableCell className="hidden md:table-cell text-sm text-muted-foreground">{inv.workOrderTitle || '—'}</TableCell>
   606→            <TableCell className="text-right font-mono">{formatINR(inv.amount)}</TableCell>
   607→            <TableCell className="hidden lg:table-cell text-right font-mono text-muted-foreground">{formatINR(inv.tax)}</TableCell>
