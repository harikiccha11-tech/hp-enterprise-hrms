   165→function PortalLoginForm({ portal, onBack }: { portal: PortalConfig; onBack: () => void }) {
   166→  const { setUser } = useAppStore()
   167→  const [username, setUsername] = useState('')
   168→  const [password, setPassword] = useState('')
   169→  const [loading, setLoading] = useState(false)
   170→  const [forgotOpen, setForgotOpen] = useState(false)
   171→  const Icon = portal.icon
   172→
   173→  const handleSubmit = async (e: React.FormEvent) => {
   174→    e.preventDefault()
   175→    if (!username.trim() || !password.trim()) {
   176→      toast.error('Please enter username and password')
   177→      return
   178→    }
   179→    setLoading(true)
   180→    try {
   181→      const res = await fetch('/api/auth/login', {
   182→        method: 'POST',
   183→        headers: { 'Content-Type': 'application/json' },
   184→        body: JSON.stringify({ username: username.trim(), password }),
   185→      })
   186→      const data = await res.json()
   187→      if (res.ok && data.user) {
   188→        setUser(data.user)
   189→        toast.success('Welcome back!')
   190→      } else {
   191→        toast.error(data.error || 'Login failed. Please check your credentials.')
   192→      }
   193→    } catch {
   194→      toast.error('Network error. Please try again.')
   195→    } finally {
   196→      setLoading(false)
   197→    }
   198→  }
   199→
   200→  return (
   201→    <>
   202→      <section className="min-h-screen flex items-center justify-center px-4 relative overflow-hidden" style={{ background: 'linear-gradient(135deg, #0a0e1a 0%, #0d1528 30%, #0f1d36 60%, #081225 100%)' }}>

   307→    <section className="min-h-screen py-20 px-4 relative overflow-hidden" style={{ background: 'linear-gradient(135deg, #0a0e1a, #0d1528)' }}>
   308→      <div className="absolute inset-0 overflow-hidden pointer-events-none" aria-hidden="true">

   313→          <Badge className="mb-4 px-4 py-1" style={{ background: 'rgba(212,175,55,0.1)', color: 'var(--gold, #D4AF37)', border: '1px solid rgba(212,175,55,0.2)' }}>Get Started</Badge>
