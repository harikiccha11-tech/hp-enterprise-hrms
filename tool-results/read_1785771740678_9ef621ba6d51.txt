   171→function Reveal({ children, delay = 0, direction = 'up', className = '' }: { children: React.ReactNode; delay?: number; direction?: 'up' | 'left' | 'right'; className?: string }) {
   172→  const ref = useRef<HTMLDivElement>(null)
   173→  const isInView = useInView(ref, { once: true, margin: '-40px' })
   174→  const offsets = { up: { y: 50, x: 0 }, left: { y: 0, x: -50 }, right: { y: 0, x: 50 } }
   175→  return (
   176→    <motion.div ref={ref} className={className}
   177→      initial={{ opacity: 0, ...offsets[direction] }}
   178→      animate={isInView ? { opacity: 1, x: 0, y: 0 } : { opacity: 0, ...offsets[direction] }}
   179→      transition={{ duration: 0.8, delay, ease: LUXURY_EASE }}
   180→    >
   181→      {children}
   182→    </motion.div>
   183→  )

   286→function PortalLoginCard({ portal, onForgotPassword }: { portal: PortalConfig; onForgotPassword: () => void }) {
   287→  const { setUser } = useAppStore()
   288→  const [username, setUsername] = useState('')
   289→  const [password, setPassword] = useState('')
   290→  const [loading, setLoading] = useState(false)
   291→  const Icon = portal.icon
   292→
   293→  const handleSubmit = async (e: React.FormEvent) => {
   294→    e.preventDefault()
   295→    if (!username.trim() || !password.trim()) { toast.error('Please enter username and password'); return }
   296→    setLoading(true)
   297→    try {
   298→      const res = await fetch('/api/auth/login', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ username: username.trim(), password }) })
   299→      const data = await res.json()
   300→      if (res.ok && data.user) { setUser(data.user); toast.success(`Welcome to ${portal.title}!`) }
   301→      else { toast.error(data.error || 'Login failed. Please check your credentials.') }
   302→    } catch { toast.error('Network error. Please try again.') } finally { setLoading(false) }
   303→  }
   304→
   305→  return (
   306→    <form onSubmit={handleSubmit} className="space-y-3">
   307→      <div className="space-y-1.5">
   308→        <Label htmlFor={`login-${portal.id}`} className="text-gray-800 text-xs font-bold uppercase tracking-wider">Username</Label>
   309→        <Input id={`login-${portal.id}`} placeholder="Enter your username" value={username} onChange={(e) => setUsername(e.target.value)} className="h-11 text-sm bg-white" autoComplete="username" />
   310→      </div>
   311→      <div className="space-y-1.5">
   312→        <div className="flex items-center justify-between">
   313→          <Label htmlFor={`pass-${portal.id}`} className="text-gray-800 text-xs font-bold uppercase tracking-wider">Password</Label>
   314→          <button type="button" onClick={onForgotPassword} className="text-[11px] font-bold hover:underline" style={{ color: portal.color }}>Forgot?</button>
   315→        </div>
   316→        <Input id={`pass-${portal.id}`} type="password" placeholder="Enter your password" value={password} onChange={(e) => setPassword(e.target.value)} className="h-11 text-sm bg-white" autoComplete="current-password" />
   317→      </div>
   318→      <Button type="submit" className="w-full h-11 font-bold text-sm text-white" style={{ background: portal.color }} disabled={loading}>
   319→        {loading ? <Loader2 className="mr-2 h-3.5 w-3.5 animate-spin" /> : <Lock className="mr-2 h-3.5 w-3.5" />}Sign In to {portal.title}
   320→      </Button>
   321→    </form>
