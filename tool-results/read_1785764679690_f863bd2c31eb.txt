   223→function Reveal({ children, className, delay = 0, direction = 'up' }: {
   224→  children: React.ReactNode; className?: string; delay?: number; direction?: 'up' | 'down' | 'left' | 'right'
   225→}) {

   230→  return (
   231→    <motion.div
   232→      ref={ref}
   233→      initial={{ opacity: 0, ...dir }}
   234→      animate={inView ? { opacity: 1, x: 0, y: 0 } : {}}
   235→      transition={{ duration: 0.6, delay, ease: [0.25, 0.4, 0.25, 1] }}
   236→      className={className}
   237→    >
   238→      {children}
   239→    </motion.div>
