   195→   195→export function Attendance({ refreshKey }: { refreshKey: number }) {
   196→   196→  const { user } = useAuth()
   197→   197→  const isSuperAdmin = user?.role === 'OWNER' || user?.role === 'SUPER_ADMIN'
   198→   198→  const isClient = user?.role === 'CLIENT'
   199→   199→  const [list, setList] = useState<AttendanceRecord[]>([])
   200→   200→  const [loading, setLoading] = useState(true)
   201→   201→  const [date, setDate] = useState('')
   202→   202→  const [clientId, setClientId] = useState<string | null>(null)
   203→   203→  const [clients, setClients] = useState<ClientOption[]>([])
   204→   204→  const [editing, setEditing] = useState<AttendanceRecord | null>(null)
   205→   205→  const [deleting, setDeleting] = useState<AttendanceRecord | null>(null)
   206→   206→  const [selfieView, setSelfieView] = useState<{ url: string; label: string } | null>(null)
   207→   207→
   208→   208→  /* ── Load clients for filter dropdown ── */
   209→   209→  useEffect(
   210→   210→    function () {
   211→   211→      if (isClient) return
   212→   212→      api<{ clients: ClientOption[] }>('/api/admin/clients?limit=100')
   213→   213→        .then(function (d) {
   214→   214→          setClients(d.clients || [])
   215→   215→        })
   216→   216→        .catch(function () {
   217→   217→          /* ignore */
   218→   218→        })
   219→   219→    },
   220→   220→    [isClient],
   221→   221→  )
   222→   222→
   223→   223→  /* ── Load attendance records ── */
   224→   224→  const load = useCallback(
   225→   225→    async function () {
   226→   226→      setLoading(true)
   227→   227→      try {
   228→   228→        const params = new URLSearchParams()
   229→   229→        if (date) params.set('date', date)
   230→   230→        if (clientId) params.set('clientId', clientId)
   231→   231→        const data = await api<{ records: AttendanceRecord[] }>(
   232→   232→          '/api/admin/attendance?' + params.toString(),
   233→   233→        )
   234→   234→        setList(data.records || [])
   235→   235→      } catch (e: any) {
   236→   236→        toast.error(e.message || 'Failed to load attendance')
   237→   237→      } finally {
   238→   238→        setLoading(false)
   239→   239→      }
   240→   240→    },
   241→   241→    [date, clientId],
   242→   242→  )
   243→   243→
   244→   244→  useEffect(
   245→   245→    function () {
   246→   246→      load()
   247→   247→    },
   248→   248→    [load, refreshKey],
   249→   249→  )
   250→   250→
   251→   251→  /* ── Named handlers (no inline arrow fns in JSX) ── */
   252→   252→  const handleClientFilterChange = function (v: string) {
   253→   253→    if (v === '_all') {
   254→   254→      setClientId(undefined)
   255→   255→    } else {
   256→   256→      setClientId(v)
   257→   257→    }
   258→   258→  }
   259→   259→
   260→   260→  const handleClearFilters = function () {
   261→   261→    setDate('')
   262→   262→    setClientId(undefined)
   263→   263→  }
   264→   264→
   265→   265→  const handleOpenSelfie = function (url: string, label: string) {
   266→   266→    setSelfieView({ url: url, label: label })
   267→   267→  }
   268→   268→
   269→   269→  const handleCloseSelfie = function () {
   270→   270→    setSelfieView(null)
   271→   271→  }
   272→   272→
   273→   273→  const handleExportCSV = function () {
   274→   274→    const esc = function (s: string) {
   275→   275→      return '"' + s.replace(/"/g, '""') + '"'
   276→   276→    }
   277→   277→    const cols = [
   278→   278→      { key: 'employee.employeeCode', label: 'Code' },
   279→   279→      { key: 'employee.fullName', label: 'Name' },
   280→   280→      { key: 'employee.assignedClient.clientName', label: 'Client' },
   281→   281→      { key: 'date', label: 'Date' },
   282→   282→      { key: 'punchIn', label: 'Punch In' },
   283→   283→      { key: 'punchOut', label: 'Punch Out' },
   284→   284→      { key: 'workingHours', label: 'Hours' },
   285→   285→      { key: 'overtime', label: 'OT' },
   286→   286→      { key: 'status', label: 'Status' },
   287→   287→    ]
   288→   288→    let csv = cols.map(function (c) {
   289→   289→      return c.label
   290→   290→    }).join(',') + '\n'
   291→   291→    for (const r of list) {
   292→   292→      const row = cols
   293→   293→        .map(function (c) {
   294→   294→          const parts = c.key.split('.')
   295→   295→          let v: any = r
   296→   296→          for (const p of parts) {
   297→   297→            v = v && v[p]
   298→   298→          }
   299→   299→          return typeof v === 'string' ? esc(v) : v != null ? String(v) : ''
   300→   300→        })
   301→   301→        .join(',')
   302→   302→      csv += row + '\n'
   303→   303→    }
   304→   304→    const blob = new Blob([csv], { type: 'text/csv' })
   305→   305→    const url = URL.createObjectURL(blob)
   306→   306→    const a = document.createElement('a')
   307→   307→    a.href = url
   308→   308→    a.download = 'attendance_' + new Date().toISOString().slice(0, 10) + '.csv'
   309→   309→    a.click()
   310→   310→    URL.revokeObjectURL(url)
   311→   311→  }
   312→   312→
   313→   313→  /* ── Pre-computed values (avoid complex expressions in JSX) ── */
   314→   314→  const trackedRecords = list.filter(function (r) {
   315→   315→    return r.punchInAddress
   316→   316→  })
   317→   317→  const trackedCount = trackedRecords.length
   318→   318→  const showLocationCards = !loading && list.length > 0 && trackedCount > 0
   319→   319→  const showClientFilter = !isClient && clients.length > 0
   320→   320→  const hasFilters = !!(date || clientId)
   321→   321→  const sectionDesc = isClient
   322→   322→    ? 'Attendance records for your assigned employees'
   323→   323→    : 'Daily punch records — filter by date and client'
   324→   324→  const deletingName = deleting ? deleting.employee.fullName : ''
   325→   325→  const deletingDate = deleting ? fmtDate(deleting.date) : ''
   326→   326→
   327→   327→  return (
   328→   328→    <div className="space-y-5">
   329→   329→      <SectionTitle
   330→   330→        title="Attendance"
   331→   331→        desc={sectionDesc}
   332→   332→        action={
   333→   333→          <div className="flex flex-wrap items-center gap-2">
   334→   334→            {list.length > 0 && (
   335→   335→              <Button variant="outline" size="sm" onClick={handleExportCSV}>
   336→   336→                <Download className="mr-1.5 h-3.5 w-3.5" /> Export CSV
   337→   337→              </Button>
   338→   338→            )}
   339→   339→            {showClientFilter && (
   340→   340→              <Select value={clientId || '_all'} onValueChange={handleClientFilterChange}>
   341→   341→                <SelectTrigger className="w-[180px] h-9">
   342→   342→                  <Building2 className="mr-1.5 h-3.5 w-3.5 text-muted-foreground" />
   343→   343→                  <SelectValue placeholder="All Clients" />
   344→   344→                </SelectTrigger>
   345→   345→                <SelectContent>
   346→   346→                  <SelectItem value="_all">All Clients</SelectItem>
   347→   347→                  {clients.map(function (c) {
   348→   348→                    return (
   349→   349→                      <SelectItem key={c.id} value={c.id}>
   350→   350→                        {c.clientName}
   351→   351→                      </SelectItem>
   352→   352→                    )
   353→   353→                  })}
   354→   354→                </SelectContent>
   355→   355→              </Select>
   356→   356→            )}
   357→   357→            <div className="relative">
   358→   358→              <Calendar className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   359→   359→              <Input
   360→   360→                type="date"
   361→   361→                value={date}
   362→   362→                onChange={function (e) {
   363→   363→                  setDate(e.target.value)
   364→   364→                }}
   365→   365→                className="pl-9 w-[180px] h-9"
   366→   366→              />
   367→   367→            </div>
   368→   368→            {hasFilters && (
   369→   369→              <Button variant="ghost" size="sm" onClick={handleClearFilters}>
   370→   370→                Clear
   371→   371→              </Button>
   372→   372→            )}
   373→   373→          </div>
   374→   374→        }
   375→   375→      />
   376→   376→
   377→   377→      {/* Live check-in locations */}
   378→   378→      {showLocationCards && (
   379→   379→        <Card className="border-[var(--gold)]/30 bg-[var(--gold)]/5">
   380→   380→          <CardContent className="pt-5 pb-5">
   381→   381→            <div className="flex items-center gap-2 mb-3">
   382→   382→              <MapPin className="h-4 w-4 text-[var(--gold)]" />
   383→   383→              <h3 className="text-sm font-bold text-[var(--navy)] dark:text-white">Live Check-in Locations</h3>
   384→   384→              <Badge variant="outline" className="border-[var(--gold)]/40 text-[#8a6f24]">
   385→   385→                {trackedCount} tracked
   386→   386→              </Badge>
   387→   387→            </div>
   388→   388→            <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-3 max-h-48 overflow-y-auto scroll-thin">
   389→   389→              {trackedRecords.map(function (r) {
   390→   390→                return <LocationCard key={r.id} record={r} />
   391→   391→              })}
   392→   392→            </div>
   393→   393→          </CardContent>
   394→   394→        </Card>
   395→   395→      )}
   396→   396→
   397→   397→      <Card>
   398→   398→        <CardContent className="pt-6">
   399→   399→          {loading ? (
   400→   400→            <div className="space-y-2">
   401→   401→              {Array.from({ length: 8 }).map(function (_, i) {
   402→   402→                return <Skeleton key={i} className="h-14 w-full" />
   403→   403→              })}
   404→   404→            </div>
   405→   405→          ) : list.length === 0 ? (
   406→   406→            <EmptyState
   407→   407→              icon={Fingerprint}
   408→   408→              title="No attendance records"
   409→   409→              desc="Try selecting a different date or client"
   410→   410→            />
   411→   411→          ) : (
   412→   412→            <div className="max-h-[65vh] overflow-y-auto scroll-thin rounded-lg border">
   413→   413→              <Table>
   414→   414→                <TableHeader className="sticky top-0 z-10 bg-card">
   415→   415→                  <TableRow>
   416→   416→                    <TableHead>Employee</TableHead>
   417→   417→                    {!isClient && (
   418→   418→                      <TableHead className="hidden lg:table-cell">Client</TableHead>
   419→   419→                    )}
   420→   420→                    <TableHead>Date</TableHead>
   421→   421→                    <TableHead>Punch In</TableHead>
   422→   422→                    <TableHead>Punch Out</TableHead>
   423→   423→                    <TableHead className="hidden md:table-cell">Hours</TableHead>
   424→   424→                    <TableHead className="hidden lg:table-cell">OT</TableHead>
   425→   425→                    <TableHead className="hidden sm:table-cell">Selfie</TableHead>
   426→   426→                    <TableHead className="hidden xl:table-cell">Location</TableHead>
   427→   427→                    <TableHead>Status</TableHead>
   428→   428→                    <TableHead className="text-right">Actions</TableHead>
   429→   429→                  </TableRow>
   430→   430→                </TableHeader>
   431→   431→                <TableBody>
   432→   432→                  {list.map(function (r) {
   433→   433→                    return (
   434→   434→                      <TableRow key={r.id} className="hover:bg-muted/40">
   435→   435→                        <TableCell>
   436→   436→                          <div className="min-w-0">
   437→   437→                            <p className="truncate text-sm font-semibold text-[var(--navy)] dark:text-white">
   438→   438→                              {r.employee.fullName}
   439→   439→                            </p>
   440→   440→                            <p className="truncate text-[11px] text-muted-foreground">
   441→   441→                              {r.employee.employeeCode || '—'} • {r.employee.department || '—'}
   442→   442→                            </p>
   443→   443→                          </div>
   444→   444→                        </TableCell>
   445→   445→                        {!isClient && (
   446→   446→                          <TableCell className="hidden lg:table-cell">
   447→   447→                            <ClientNameCell record={r} />
   448→   448→                          </TableCell>
   449→   449→                        )}
   450→   450→                        <TableCell className="text-sm whitespace-nowrap">{fmtDate(r.date)}</TableCell>
   451→   451→                        <TableCell className="text-sm whitespace-nowrap">
   452→   452→                          {r.punchIn ? fmtTime(r.punchIn) : '—'}
   453→   453→                        </TableCell>
   454→   454→                        <TableCell className="text-sm whitespace-nowrap">
   455→   455→                          {r.punchOut ? fmtTime(r.punchOut) : '—'}
   456→   456→                        </TableCell>
   457→   457→                        <TableCell className="hidden md:table-cell text-sm">
   458→   458→                          {r.workingHours ? r.workingHours + 'h' : '—'}
   459→   459→                        </TableCell>

   613→   613→/* ── Edit dialog (separate component, no Turbopack issue) ── */
   614→   614→function EditAttendanceDialog(props: {
   615→   615→  record: AttendanceRecord
   616→   616→  onClose: () => void
   617→   617→  onSuccess: () => void
   618→   618→}) {
   619→   619→  const { record, onClose, onSuccess } = props
   620→   620→  const toLocalInput = function (d: string | null) {
   621→   621→    if (!d) return ''
   622→   622→    try {
   623→   623→      const dt = new Date(d)
   624→   624→      const pad = function (n: number) {
   625→   625→        return String(n).padStart(2, '0')
   626→   626→      }
   627→   627→      return (
   628→   628→        dt.getFullYear() +
   629→   629→        '-' +
   630→   630→        pad(dt.getMonth() + 1) +
   631→   631→        '-' +
   632→   632→        pad(dt.getDate()) +
   633→   633→        'T' +
   634→   634→        pad(dt.getHours()) +
   635→   635→        ':' +
   636→   636→        pad(dt.getMinutes())
   637→   637→      )
   638→   638→    } catch {
   639→   639→      return ''
   640→   640→    }
   641→   641→  }
   642→   642→
   643→   643→  const [punchIn, setPunchIn] = useState(toLocalInput(record.punchIn))
   644→   644→  const [punchOut, setPunchOut] = useState(toLocalInput(record.punchOut))
   645→   645→  const [status, setStatus] = useState(record.status)
   646→   646→  const [saving, setSaving] = useState(false)
   647→   647→
   648→   648→  const submit = async function () {
   649→   649→    setSaving(true)
   650→   650→    try {
   651→   651→      await api('/api/admin/attendance', {
   652→   652→        method: 'PATCH',
   653→   653→        body: JSON.stringify({
   654→   654→          id: record.id,
   655→   655→          punchIn: punchIn ? new Date(punchIn).toISOString() : undefined,
   656→   656→          punchOut: punchOut ? new Date(punchOut).toISOString() : undefined,
   657→   657→          status,
   658→   658→        }),
   659→   659→      })
   660→   660→      toast.success('Attendance updated')
   661→   661→      onSuccess()
   662→   662→    } catch (e: any) {
   663→   663→      toast.error(e.message || 'Update failed')
   664→   664→    } finally {
   665→   665→      setSaving(false)
   666→   666→    }
   667→   667→  }
   668→   668→
   669→   669→  return (
   670→   670→    <Dialog open onOpenChange={function (o) { if (!o) onClose() }}>
   671→   671→      <DialogContent className="max-w-md">
   672→   672→        <DialogHeader>
   673→   673→          <DialogTitle className="flex items-center gap-2">
   674→   674→            <Pencil className="h-5 w-5" /> Edit Attendance
   675→   675→          </DialogTitle>
   676→   676→          <DialogDescription>
   677→   677→            {record.employee.fullName} • {fmtDate(record.date)}
   678→   678→          </DialogDescription>
   679→   679→        </DialogHeader>
   680→   680→        <div className="space-y-3">
   681→   681→          <div className="space-y-1.5">
   682→   682→            <Label htmlFor="at-in">Punch In</Label>
   683→   683→            <Input
   684→   684→              id="at-in"
   685→   685→              type="datetime-local"
   686→   686→              value={punchIn}
   687→   687→              onChange={function (e) {
   688→   688→                setPunchIn(e.target.value)
   689→   689→              }}
   690→   690→            />
   691→   691→          </div>
   692→   692→          <div className="space-y-1.5">
   693→   693→            <Label htmlFor="at-out">Punch Out</Label>
   694→   694→            <Input
   695→   695→              id="at-out"
   696→   696→              type="datetime-local"
   697→   697→              value={punchOut}
   698→   698→              onChange={function (e) {
   699→   699→                setPunchOut(e.target.value)
   700→   700→              }}
   701→   701→            />
   702→   702→          </div>
   703→   703→          <div className="space-y-1.5">
   704→   704→            <Label htmlFor="at-status">Status</Label>
   705→   705→            <Select value={status} onValueChange={setStatus}>
   706→   706→              <SelectTrigger id="at-status">
   707→   707→                <SelectValue />
   708→   708→              </SelectTrigger>
   709→   709→              <SelectContent>
   710→   710→                {STATUS_OPTIONS.map(function (s) {
   711→   711→                  return (
   712→   712→                    <SelectItem key={s.value} value={s.value}>
   713→   713→                      {s.label}
   714→   714→                    </SelectItem>
   715→   715→                  )
   716→   716→                })}
   717→   717→              </SelectContent>
   718→   718→            </Select>
   719→   719→          </div>
   720→   720→        </div>
   721→   721→        <DialogFooter>
   722→   722→          <Button variant="outline" onClick={onClose}>
   723→   723→            Cancel
   724→   724→          </Button>
   725→   725→          <Button
   726→   726→            className="bg-[var(--navy)] text-white hover:bg-[var(--navy-light)]"
   727→   727→            onClick={submit}

   729→   729→          >
   730→   730→            {saving ? 'Saving…' : 'Save'}
   731→   731→          </Button>
   732→   732→        </DialogFooter>
   733→   733→      </DialogContent>
   734→   734→    </Dialog>
   735→   735→  )
