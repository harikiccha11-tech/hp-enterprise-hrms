    97→    97→function statusColor(status: string) {
    98→    98→  if (status === 'ACTIVE') return 'bg-emerald-500/10 text-emerald-700 border-emerald-500/30'
    99→    99→  if (status === 'REVOKED' || status === 'EXPIRED') return 'bg-red-500/10 text-red-700 border-red-500/30'
   100→   100→  return 'bg-gray-500/10 text-gray-700 border-gray-500/30'
   101→   101→}
   102→   102→
   103→   103→function maskKey(key: string) {
   104→   104→  return key.slice(0, 12) + '••••••••••••••••' + key.slice(-4)
   105→   105→}
   106→   106→
   107→   107→function formatDate(iso: string) {
   108→   108→  return new Date(iso).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
   109→   109→}
   110→   110→
   111→   111→function relativeTime(iso: string) {
   112→   112→  const diff = Date.now() - new Date(iso).getTime()
   113→   113→  const mins = Math.floor(diff / 60000)
   114→   114→  if (mins < 60) return `${mins}m ago`
   115→   115→  const hours = Math.floor(mins / 60)
   116→   116→  if (hours < 24) return `${hours}h ago`
   117→   117→  const days = Math.floor(hours / 24)
   118→   118→  if (days < 30) return `${days}d ago`
   119→   119→  return formatDate(iso)
   120→   120→}
