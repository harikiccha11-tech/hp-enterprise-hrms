    68→// ── Portal Login Form ──────────────────────────────────────────────────────
    69→
    70→function PortalLoginForm({ portal, onBack }: { portal: typeof PORTALS[number]; onBack: () => void }) {
    71→  const [username, setUsername] = useState('')
    72→  const [password, setPassword] = useState('')
    73→  const [loading, setLoading] = useState(false)
    74→  const { setUser } = useAppStore()
    75→  const Icon = portal.icon
    76→
    77→  async function handleLogin(e: React.FormEvent) {
    78→    e.preventDefault()
    79→    setLoading(true)
    80→    try {
    81→      const res = await fetch('/api/auth/login', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ username, password }) })
    82→      const data = await res.json()
    83→      if (!res.ok) { toast.error(data.error || 'Invalid credentials'); return }
    84→      setUser(data.user)
    85→      toast.success('Welcome back, ' + data.user.username + '!')
    86→    } catch { toast.error('Network error') } finally { setLoading(false) }
    87→  }
    88→
    89→  return (
    90→    <div className="min-h-screen flex flex-col items-center justify-center p-4 bg-background">
    91→      <div className="w-full max-w-md">
    92→        <button onClick={onBack} className="flex items-center gap-1.5 text-sm text-muted-foreground hover:text-foreground mb-6 transition-colors">
    93→          <ArrowRight className="h-4 w-4 rotate-180" /> Back to HPHRMS
    94→        </button>
    95→        <Card className="border-0 shadow-xl">
    96→          <CardHeader className="text-center pb-2">
    97→            <div className="mx-auto grid h-14 w-14 place-items-center rounded-2xl" style={{ backgroundColor: portal.color }}>
    98→              <Icon className="h-7 w-7 text-white" />
    99→            </div>
   100→            <CardTitle className="text-xl mt-3">{portal.title}</CardTitle>
   101→            <CardDescription>{portal.subtitle}</CardDescription>
   102→          </CardHeader>
   103→          <CardContent>
   104→            <form onSubmit={handleLogin} className="space-y-4">
   105→              <div className="space-y-1.5">
   106→                <Label>Username</Label>
   107→                <div className="relative">
   108→                  <UserIcon className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   109→                  <Input value={username} onChange={(e) => setUsername(e.target.value)} placeholder="Enter username" className="pl-9" autoComplete="username" autoFocus />
   110→                </div>
   111→              </div>
   112→              <div className="space-y-1.5">
   113→                <div className="flex items-center justify-between"><Label>Password</Label><ForgotPasswordDialog /></div>
   114→                <div className="relative">
   115→                  <Lock className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   116→                  <Input type="password" value={password} onChange={(e) => setPassword(e.target.value)} placeholder="Enter password" className="pl-9" autoComplete="current-password" />
   117→                </div>
   118→              </div>
   119→              <Button type="submit" disabled={loading} className="w-full text-white" style={{ backgroundColor: portal.color }}>
   120→                {loading ? <><Loader2 className="mr-2 h-4 w-4 animate-spin" /> Signing in...</> : <>Sign In to {portal.title}</>}
   121→              </Button>
   122→            </form>
   123→            <div className="mt-4 rounded-lg border border-dashed p-3 text-center">
   124→              <p className="text-[10px] text-muted-foreground uppercase tracking-wider mb-1">Demo Credentials</p>

   126→            </div>
   127→          </CardContent>
   128→        </Card>
   129→      </div>
   130→    </div>
