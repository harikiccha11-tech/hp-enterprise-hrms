   115→function initials(name: string): string {

   132→function statusColor(status: string): string {
   133→  const s = status.toLowerCase()
   134→  if (s === 'active' || s === 'in progress' || s === 'paid') return 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 border-emerald-500/25'
   135→  if (s === 'pending' || s === 'submitted') return 'bg-amber-500/15 text-amber-700 dark:text-amber-400 border-amber-500/25'
   136→  if (s === 'completed' || s === 'closed' || s === 'approved') return 'bg-sky-500/15 text-sky-700 dark:text-sky-400 border-sky-500/25'
   137→  if (s === 'cancelled' || s === 'rejected' || s === 'overdue') return 'bg-red-500/15 text-red-700 dark:text-red-400 border-red-500/25'
   138→  if (s === 'draft' || s === 'on hold') return 'bg-slate-500/15 text-slate-700 dark:text-slate-400 border-slate-500/25'
   139→  return 'bg-primary/10 text-primary border-primary/20'
   140→}
