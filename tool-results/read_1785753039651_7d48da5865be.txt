   200→interface NotificationLite {
   201→  id: string
   202→  title: string
   203→  message: string
   204→  type: string
   205→  read: boolean
   206→  createdAt: string
   207→  link?: string | null
   208→}

   210→export function AdminLayout() {
   211→  const { user, logout, lang } = useAuth()
   212→  const [active, setActive] = useState<ModuleKey>('dashboard')
   213→  const [mobileOpen, setMobileOpen] = useState(false)
   214→  const [notifications, setNotifications] = useState<NotificationLite[]>([])
   215→  const [unread, setUnread] = useState(0)
   216→  const [now, setNow] = useState(new Date())
   217→  const [bellOpen, setBellOpen] = useState(false)
   218→  const [refreshKey, setRefreshKey] = useState(0)
   219→  const sseRef = useRef<EventSource | null>(null)
   220→
   221→  const isOwner = user?.role === 'OWNER'
   222→  const isSuperAdmin = user?.role === 'OWNER' || user?.role === 'SUPER_ADMIN'
   223→  const roleLabel = user?.role ? ROLE_LABELS[user.role] : 'Admin'
   224→  const adminName = user?.username || 'Admin'
   225→  const adminEmail = user?.email || ''
   226→
   227→  useEffect(() => {
   228→    let cancelled = false
   229→    fetch('/api/notifications', { cache: 'no-store' })
   230→      .then((r) => (r.ok ? r.json() : null))
   231→      .then((data) => {
   232→        if (cancelled || !data) return
   233→        setNotifications(data.notifications || [])
   234→        setUnread(data.unread || 0)
   235→      })
   236→      .catch(() => {})
   237→    return () => { cancelled = true }
   238→  }, [])
   239→
   240→  useEffect(() => {
   241→    const t = setInterval(() => setNow(new Date()), 30_000)
   242→    return () => clearInterval(t)
   243→  }, [])
   244→
   245→  useEffect(() => {
   246→    let es: EventSource | null = null
   247→    try {
   248→      es = new EventSource('/api/sse')
   249→      sseRef.current = es
   250→      es.onmessage = (ev) => {
   251→        try {
   252→          const payload = JSON.parse(ev.data)
   253→          if (payload.type === 'notification') {
   254→            toast(payload.notification.title, { description: payload.notification.message })
   255→            fetch('/api/notifications', { cache: 'no-store' })
   256→              .then((r) => r.ok ? r.json() : null)
   257→              .then((d) => { if (!d) return; setNotifications(d.notifications || []); setUnread(d.unread || 0) })
   258→              .catch(() => {})
   259→            setRefreshKey((k) => k + 1)
   260→          }
   261→        } catch {}
   262→      }
   263→      es.onerror = () => {}
   264→    } catch {}
   265→    return () => { try { es?.close() } catch {} }
   266→  }, [])
   267→
   268→  const loadNotifications = async () => {
   269→    try {
   270→      const res = await fetch('/api/notifications', { cache: 'no-store' })
   271→      if (!res.ok) return
   272→      const data = await res.json()
   273→      setNotifications(data.notifications || [])
   274→      setUnread(data.unread || 0)
   275→    } catch {}
   276→  }
   277→
   278→  const handleNavigate = (k: ModuleKey) => { setActive(k); setMobileOpen(false) }
   279→
   280→  const markAllRead = async () => {
   281→    try {
   282→      await fetch('/api/notifications', { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ all: true }) })
   283→      setNotifications((arr) => arr.map((n) => ({ ...n, read: true })))
   284→      setUnread(0)
   285→      toast.success(t('nav.allMarkedRead', lang))
   286→    } catch { toast.error(t('nav.notifFailed', lang)) }
   287→  }
   288→
   289→  const markOneRead = async (id: string) => {
   290→    try {
   291→      await fetch('/api/notifications', { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id }) })
   292→      setNotifications((arr) => arr.map((n) => (n.id === id ? { ...n, read: true } : n)))
   293→      setUnread((u) => Math.max(0, u - 1))
   294→    } catch {}
   295→  }
   296→
   297→  const handleLogout = async () => { await logout(); toast.success(t('nav.signedOut', lang)) }
   298→
   299→  const navGroups = getNavGroups(lang)
   300→  const allItems = navGroups.flatMap((g) => g.items)
   301→  const currentNav = allItems.find((n) => n.key === active)
   302→
   303→  const sidebarInner = (
   304→    <div className="flex h-full flex-col text-slate-200">
   305→      <div className="px-5 pt-5 pb-4 border-b border-white/10">
   306→        <BrandLogo />
   307→        <div className="mt-3 flex items-center gap-2">
   308→          <span className={cn('inline-flex items-center gap-1.5 rounded-full px-2.5 py-1 text-[11px] font-bold uppercase tracking-wider', isOwner ? 'bg-gradient-to-r from-[var(--gold)] to-[var(--gold-light)] text-[var(--navy)]' : isSuperAdmin ? 'bg-[var(--gold)]/80 text-[var(--navy)]' : 'bg-white/10 text-[var(--gold-light)]')}>
   309→            {isOwner ? <Crown className="h-3 w-3" /> : isSuperAdmin ? <ShieldCheck className="h-3 w-3" /> : <ShieldAlert className="h-3 w-3" />}
   310→            {roleLabel}
   311→          </span>
   312→          <span className="text-[10px] uppercase tracking-wider text-blue-200/60">Console</span>
   313→        </div>
   314→      </div>
   315→
   316→      <nav className="flex-1 overflow-y-auto scroll-thin px-3 py-3 space-y-4" aria-label="Admin navigation">
   317→        {navGroups.map((group) => {
   318→          const visibleItems = group.items.filter((it) => (!it.superAdminOnly || isSuperAdmin) && (!it.ownerOnly || isOwner))
   319→          if (visibleItems.length === 0) return null
   320→          return (
   321→            <div key={group.title}>
   322→              <p className="px-3 pb-1.5 text-[10px] font-semibold uppercase tracking-[0.18em] text-blue-200/50">{group.title}</p>
   323→              <div className="space-y-1">
   324→                {visibleItems.map((item) => {
   325→                  const Icon = item.icon
   326→                  const isActive = active === item.key
   327→                  return (
   328→                    <button key={item.key} onClick={() => handleNavigate(item.key)} aria-current={isActive ? 'page' : undefined}
   329→                      className={cn('group flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-left text-sm font-medium transition-all', isActive ? 'bg-[var(--gold)] text-[var(--navy)] shadow-sm' : 'text-blue-100/80 hover:bg-white/10 hover:text-white')}>
   330→                      <Icon className={cn('h-[18px] w-[18px] shrink-0', isActive ? 'text-[var(--navy)]' : 'text-blue-200/70 group-hover:text-[var(--gold-light)]')} />
   331→                      <span className="flex-1 truncate">{item.label}</span>
   332→                      {isActive && <ChevronRight className="h-4 w-4 text-[var(--navy)]" />}
   333→                    </button>
   334→                  )
   335→                })}
   336→              </div>
   337→            </div>
   338→          )
   339→        })}
   340→      </nav>
   341→
   342→      <div className="border-t border-white/10 p-3">
   343→        <div className="rounded-lg bg-white/5 p-3">
   344→          <div className="flex items-center gap-3">
   345→            <Avatar className="h-9 w-9 ring-2 ring-[var(--gold)]/40">
   346→              <AvatarFallback className="bg-[var(--gold)] text-[var(--navy)] text-xs font-bold">{initials(adminName)}</AvatarFallback>
   347→            </Avatar>
   348→            <div className="min-w-0 flex-1">
   349→              <p className="truncate text-sm font-semibold text-white">{adminName}</p>
   350→              <p className="truncate text-[11px] text-blue-200/70">{adminEmail || roleLabel}</p>
   351→            </div>
   352→          </div>
   353→          <Button onClick={handleLogout} variant="ghost" size="sm" className="mt-3 w-full justify-start gap-2 text-blue-100/80 hover:bg-white/10 hover:text-white">
   354→            <LogOut className="h-4 w-4" /> {t('nav.signOut', lang)}
   355→          </Button>
   356→        </div>
   357→      </div>
   358→    </div>
   359→  )
   360→
   361→  return (
   362→    <div className="min-h-screen flex flex-col bg-background">
   363→      <div className="h-1 hpe-gold-bar w-full" />
   364→      <div className="flex flex-1 overflow-hidden">
   365→        <aside className="hidden lg:flex w-[260px] shrink-0 hpe-sidebar-gradient border-r border-white/10">{sidebarInner}</aside>
   366→        <Sheet open={mobileOpen} onOpenChange={setMobileOpen}>
   367→          <SheetContent side="left" className="w-[260px] p-0 hpe-sidebar-gradient border-white/10">
   368→            <SheetTitle className="sr-only">HP ENTERPRISE Admin Navigation</SheetTitle>
   369→            <SheetDescription className="sr-only">Admin console navigation menu.</SheetDescription>
   370→            <button onClick={() => setMobileOpen(false)} className="absolute right-3 top-3 z-10 grid h-8 w-8 place-items-center rounded-md text-blue-100 hover:bg-white/10" aria-label="Close menu"><X className="h-5 w-5" /></button>
   371→            {sidebarInner}
   372→          </SheetContent>
   373→        </Sheet>
   374→
   375→        <div className="flex flex-1 flex-col min-w-0">
   376→          <header className="sticky top-0 z-30 flex h-16 items-center justify-between gap-3 border-b bg-card/95 px-4 backdrop-blur-md sm:px-6">
   377→            <div className="flex items-center gap-3 min-w-0">
   378→              <button onClick={() => setMobileOpen(true)} className="lg:hidden grid h-9 w-9 place-items-center rounded-md border bg-background" aria-label="Open menu"><Menu className="h-5 w-5" /></button>
   379→              <div className="min-w-0">
   380→                <h1 className="truncate text-base font-bold text-[var(--navy)] dark:text-white sm:text-lg">{currentNav?.label}</h1>
   381→                <p className="hidden text-xs text-muted-foreground sm:block">{currentNav?.desc}</p>
   382→              </div>
   383→              {currentNav?.superAdminOnly && (<Badge variant="outline" className="hidden md:inline-flex border-[var(--gold)]/40 text-[#8a6f24] gap-1"><Lock className="h-3 w-3" /> Super Admin</Badge>)}
   384→            </div>
   385→            <div className="flex items-center gap-2 sm:gap-3">
   386→              <LanguageSwitcher />
   387→              <div className="hidden md:flex flex-col items-end leading-tight">
   388→                <span className="text-xs font-medium text-[var(--navy)] dark:text-white">{format(now, 'EEEE, dd MMM yyyy')}</span>
   389→                <span className="text-[11px] text-muted-foreground">{format(now, 'h:mm a')}</span>
   390→              </div>
   391→              <DropdownMenu open={bellOpen} onOpenChange={setBellOpen}>
   392→                <DropdownMenuTrigger asChild>
   393→                  <button className="relative grid h-10 w-10 place-items-center rounded-full border bg-background hover:bg-muted transition-colors" aria-label={`Notifications${unread ? `, ${unread} unread` : ''}`}>
   394→                    <Bell className="h-[18px] w-[18px] text-[var(--navy)] dark:text-white" />
   395→                    {unread > 0 && (<span className="absolute -right-0.5 -top-0.5 grid h-5 min-w-5 place-items-center rounded-full bg-[var(--gold)] px-1 text-[10px] font-bold text-[var(--navy)] ring-2 ring-card">{unread > 9 ? '9+' : unread}</span>)}
   396→                  </button>
   397→                </DropdownMenuTrigger>
   398→                <DropdownMenuContent align="end" className="w-80 p-0">
   399→                  <div className="flex items-center justify-between border-b px-3 py-2.5">
   400→                    <p className="text-sm font-semibold text-[var(--navy)] dark:text-white">{t('nav.notifications', lang)}</p>
   401→                    {unread > 0 && (<Button variant="ghost" size="sm" className="h-7 text-xs" onClick={markAllRead}><CheckCheck className="mr-1 h-3.5 w-3.5" /> {t('nav.markAllRead', lang)}</Button>)}
   402→                  </div>
   403→                  <div className="max-h-80 overflow-y-auto scroll-thin">
   404→                    {notifications.length === 0 ? (
   405→                      <div className="py-8 text-center text-sm text-muted-foreground"><Bell className="mx-auto mb-2 h-6 w-6 opacity-50" />{t('nav.noNotifications', lang)}</div>
   406→                    ) : (
   407→                      notifications.slice(0, 12).map((n) => (
   408→                        <DropdownMenuItem key={n.id} className="flex flex-col items-start gap-0.5 px-3 py-2.5 focus:bg-muted/60" onClick={() => { if (!n.read) markOneRead(n.id) }}>
   409→                          <div className="flex w-full items-start gap-2">
