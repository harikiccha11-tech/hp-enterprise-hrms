   135→function initials(name: string): string {
   136→  return name
   137→    .split(' ')
   138→    .filter(Boolean)
   139→    .slice(0, 2)
   140→    .map((w) => w[0]?.toUpperCase())
   141→    .join('')
   142→}
   143→
   144→function formatINR(amount: number): string {
   145→  return new Intl.NumberFormat('en-IN', {
   146→    style: 'currency',
   147→    currency: 'INR',
   148→    maximumFractionDigits: 0,
   149→  }).format(amount)
   150→}
   151→
   152→function fmtDate(d: string | null | undefined): string {
   153→  if (!d) return '—'
   154→  try {
   155→    return format(new Date(d), 'dd MMM yyyy')
   156→  } catch {
   157→    return '—'
   158→  }
   159→}
   160→
   161→function statusColor(status: string): string {
   162→  const s = status.toUpperCase()
   163→  if (s === 'ACTIVE' || s === 'OPEN' || s === 'PAID') return 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 border-emerald-500/25'
   164→  if (s === 'COMPLETED' || s === 'SENT') return 'bg-sky-500/15 text-sky-700 dark:text-sky-400 border-sky-500/25'
   165→  if (s === 'ON_HOLD' || s === 'DRAFT') return 'bg-amber-500/15 text-amber-700 dark:text-amber-400 border-amber-500/25'
   166→  if (s === 'REJECTED' || s === 'OVERDUE') return 'bg-red-500/15 text-red-700 dark:text-red-400 border-red-500/25'
   167→  if (s === 'CLOSED') return 'bg-slate-500/15 text-slate-600 dark:text-slate-400 border-slate-500/25'
   168→  return 'bg-primary/10 text-primary border-primary/20'
   169→}

   222→function DataTable<T extends { id: string }>({
   223→  title,
   224→  data,
   225→  loading,
   226→  emptyMessage,
   227→  emptyIcon: EmptyIcon,
   228→  columns,
   229→  renderRow,
   230→}: {
   231→  title: string
   232→  data: T[]
   233→  loading: boolean
   234→  emptyMessage: string
   235→  emptyIcon: typeof LayoutDashboard
   236→  columns: { key: string; label: string; className?: string }[]
   237→  renderRow: (item: T) => React.ReactNode
   238→}) {
   239→  return (
   240→    <Card>
   241→      <CardHeader className="pb-3">
   242→        <CardTitle className="text-base">{title}</CardTitle>
   243→      </CardHeader>
   244→      <CardContent className="px-6 pb-6 pt-0">
   245→        {loading ? (
   246→          <div className="space-y-3">
   247→            {Array.from({ length: 4 }).map((_, i) => (
   248→              <Skeleton key={i} className="h-10 w-full" />
   249→            ))}
   250→          </div>
   251→        ) : data.length === 0 ? (
   252→          <div className="flex flex-col items-center justify-center py-10 text-center text-muted-foreground">
   253→            <EmptyIcon className="mb-2 h-8 w-8 opacity-40" />
   254→            <p className="text-sm font-medium">{emptyMessage}</p>
   255→            <p className="mt-0.5 text-xs">Data will appear here once available</p>
   256→          </div>
   257→        ) : (
   258→          <div className="max-h-96 overflow-y-auto rounded-lg border scroll-thin">
   259→            <Table>
   260→              <TableHeader className="bg-muted/50">
   261→                <TableRow>
   262→                  {columns.map((col) => (
   263→                    <TableHead key={col.key} className={cn('text-xs font-semibold uppercase tracking-wide text-muted-foreground', col.className)}>
   264→                      {col.label}
   265→                    </TableHead>
   266→                  ))}
   267→                </TableRow>
   268→              </TableHeader>
   269→              <TableBody>
   270→                {data.map((item) => renderRow(item))}
   271→              </TableBody>
   272→            </Table>
   273→          </div>
   274→        )}
   275→      </CardContent>
   276→    </Card>
   277→  )

   637→  const [unread, setUnread] = useState(0)
   638→  const [now, setNow] = useState(new Date())
   639→  const [bellOpen, setBellOpen] = useState(false)
   640→
   641→  const client = user?.client
   642→  const clientName = data?.client?.clientName || client?.clientName || client?.companyName || user?.username || 'Client'
   643→

   648→      .then((r) => (r.ok ? r.json() : null))
   649→      .then((json) => {

   666→      .then((data) => {
   667→        if (cancelled || !data) return

   687→      await fetch('/api/notifications', {

   689→        headers: { 'Content-Type': 'application/json' },
   690→        body: JSON.stringify({ all: true }),

   692→      setNotifications((arr) => arr.map((n) => ({ ...n, read: true })))

   703→        method: 'PATCH',

   719→
   720→  // ── Sidebar inner (shared between desktop & mobile) ──

   739→                'group flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-left text-sm font-medium transition-all',
   740→                isActive

   752→              {isActive && <ChevronRight className="h-4 w-4 text-[var(--navy)]" />}

   758→      {/* User card + logout */}

   795→          {sidebarInner}

   797→
   798→        {/* Mobile sidebar (Sheet) */}
