   106→// ── Animated Counter Hook ───────────────────────────────────────────────────
   107→
   108→function useAnimatedCounter(target: number, suffix: string, startCounting: boolean, duration = 2000) {
   109→  const [count, setCount] = useState(0)
   110→  const hasAnimated = useRef(false)
   111→
   112→  useEffect(() => {
   113→    if (!startCounting || hasAnimated.current) return
   114→    hasAnimated.current = true
   115→    const startTime = Date.now()
   116→    const isDecimal = target % 1 !== 0
   117→
   118→    function tick() {
   119→      const elapsed = Date.now() - startTime
   120→      const progress = Math.min(elapsed / duration, 1)
   121→      const eased = 1 - Math.pow(1 - progress, 3)
   122→      const current = eased * target
   123→      setCount(isDecimal ? parseFloat(current.toFixed(1)) : Math.floor(current))
   124→      if (progress < 1) requestAnimationFrame(tick)
   125→    }
   126→    requestAnimationFrame(tick)
   127→  }, [startCounting, target, duration])
   128→
   129→  return `${count}${suffix}`
   130→}
