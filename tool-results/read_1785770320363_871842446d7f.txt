   224→function Reveal({ children, delay = 0, direction = 'up', className = '' }: { children: React.ReactNode; delay?: number; direction?: 'up' | 'left' | 'right'; className?: string }) {
   225→  const ref = useRef<HTMLDivElement>(null)
   226→  const isInView = useInView(ref, { once: true, margin: '-60px' })
   227→  const dirMap = { up: { y: 40 }, left: { x: -40 }, right: { x: 40 } }
   228→  return (
   229→    <motion.div ref={ref} className={className}
   230→      initial={{ opacity: 0, ...dirMap[direction] }}
   231→      animate={isInView ? { opacity: 1, x: 0, y: 0 } : { opacity: 0, ...dirMap[direction] }}
   232→      transition={{ duration: 0.6, delay, ease: [0.25, 0.4, 0.25, 1] }}
   233→    >
   234→      {children}
   235→    </motion.div>
   236→  )
   237→}

   371→function PortalLoginCard({ portal, onForgotPassword }: { portal: PortalConfig; onForgotPassword: () => void }) {
   372→  const { setUser } = useAppStore()
   373→  const [username, setUsername] = useState('')
   374→  const [password, setPassword] = useState('')
   375→  const [loading, setLoading] = useState(false)
   376→  const Icon = portal.icon
   377→
   378→  const handleSubmit = async (e: React.FormEvent) => {
   379→    e.preventDefault()
   380→    if (!username.trim() || !password.trim()) { toast.error('Please enter username and password'); return }
   381→    setLoading(true)
   382→    try {
   383→      const res = await fetch('/api/auth/login', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ username: username.trim(), password }) })
   384→      const data = await res.json()

   386→      else { toast.error(data.error || 'Login failed. Please check your credentials.') }
   387→    } catch { toast.error('Network error. Please try again.') } finally { setLoading(false) }
   388→  }
   389→
   390→  return (
   391→    <form onSubmit={handleSubmit} className="space-y-3">
   392→      <div className="space-y-1.5">
   393→        <Label htmlFor={`login-${portal.id}`} className="text-gray-800 text-xs font-semibold uppercase tracking-wider">Username</Label>

   395→      </div>
   396→      <div className="space-y-1.5">
   397→        <div className="flex items-center justify-between">
