   172→function Reveal({ children, delay = 0, direction = 'up', className = '' }: { children: React.ReactNode; delay?: number; direction?: 'up' | 'left' | 'right'; className?: string }) {
   173→  const ref = useRef<HTMLDivElement>(null)
   174→  const isInView = useInView(ref, { once: true, margin: '-40px' })
   175→  const offsets = { up: { y: 50, x: 0 }, left: { y: 0, x: -50 }, right: { y: 0, x: 50 } }
   176→  return (
   177→    <motion.div ref={ref} className={className}
   178→      initial={{ opacity: 0, ...offsets[direction] }}
   179→      animate={isInView ? { opacity: 1, x: 0, y: 0 } : { opacity: 0, ...offsets[direction] }}
   180→      transition={{ duration: 0.8, delay, ease: LUXURY_EASE }}
   181→    >
   182→      {children}
   183→    </motion.div>
   184→  )
   185→}
